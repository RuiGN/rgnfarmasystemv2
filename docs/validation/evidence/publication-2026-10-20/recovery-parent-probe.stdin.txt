"""Independent readback of authorized backup/export; no restore or service changes."""
from datetime import datetime, UTC
import gzip
import hashlib
import json
import os
from pathlib import Path
import re
import stat
import subprocess

DIR=Path('/var/backups/rgnfarmasystem/releases/prepublish-20261020T104215904590Z-63af02ef')
EXPECTED_RECEIPT='4041038ba43ea31f461cc1e8edee510201f8a5b069b06aee2d0923ce461c9f12'
IMAGE='sha256:1afda8b358fa7e8f6b839607ef8398c59d293fe4afacf797cee606c2dce9486e'
TAG='rgnfarmasystem:rollback-20261020t104215904590z-1afda8b358fa'
assert os.geteuid()==1000
assert not DIR.is_symlink() and stat.S_IMODE(DIR.stat().st_mode)==0o700
raw=(DIR/'receipt.json').read_bytes()
assert hashlib.sha256(raw).hexdigest()==EXPECTED_RECEIPT
receipt=json.loads(raw)
allowed={a['name']:a for a in receipt['artifacts']}
assert set(allowed)=={'metadata-before.json','metadata-after.json','postgresql.dump.gz','media.tar.gz','rollback-image.tar.gz'}
manifest=(DIR/'SHA256SUMS').read_text().splitlines()
found=set()
for line in manifest:
    match=re.fullmatch(r'([0-9a-f]{64})\s+\*?(.+)',line)
    assert match
    digest,name=match.groups()
    path=Path(name)
    assert '..' not in path.parts
    if not path.is_absolute():
        assert len(path.parts)==1
        path=DIR/path
    assert path.parent==DIR and path.name in allowed and not path.is_symlink()
    assert digest==allowed[path.name]['sha256']
    h=hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda:stream.read(1024*1024),b''):
            h.update(block)
    assert h.hexdigest()==digest
    assert path.stat().st_size==allowed[path.name]['bytes']
    found.add(path.name)
assert found==set(allowed)
files=[p for p in DIR.iterdir() if p.is_file() and not p.is_symlink()]
assert len(files)==7 and all(stat.S_IMODE(p.stat().st_mode)==0o600 and p.stat().st_uid==1000 for p in files)
assert not list(DIR.glob('*.partial'))
checks=[]
for name in ['postgresql.dump.gz','media.tar.gz','rollback-image.tar.gz']:
    p=subprocess.run(['gzip','-t','--',str(DIR/name)],stdout=subprocess.DEVNULL,stderr=subprocess.PIPE,timeout=90)
    checks.append({'check':'gzip','file':name,'exit_status':p.returncode})
    assert p.returncode==0
for name in ['media.tar.gz','rollback-image.tar.gz']:
    p=subprocess.run(['tar','-tzf',str(DIR/name)],stdout=subprocess.DEVNULL,stderr=subprocess.PIPE,timeout=90)
    checks.append({'check':'tar-list','file':name,'exit_status':p.returncode})
    assert p.returncode==0
producer=subprocess.Popen(['gzip','-dc','--',str(DIR/'postgresql.dump.gz')],stdout=subprocess.PIPE,stderr=subprocess.DEVNULL)
consumer=subprocess.Popen(['docker','exec','-i','rgnfarmasystem-db-1','pg_restore','--list'],stdin=producer.stdout,stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL)
producer.stdout.close()
consumer_status=consumer.wait(timeout=30)
producer_status=producer.wait(timeout=30)
assert producer_status==consumer_status==0
checks.append({'check':'pg_restore-list-no-database-connection','producer_exit':producer_status,'consumer_exit':consumer_status})
image=subprocess.run(['docker','image','inspect','--format','{{.Id}}',TAG],capture_output=True,text=True,check=True,timeout=20).stdout.strip()
assert image==IMAGE
states=subprocess.run(['docker','ps','--all','--filter','label=com.docker.compose.project=rgnfarmasystem','--format','{{.ID}}'],capture_output=True,text=True,check=True,timeout=20).stdout.splitlines()
current=[]
for identifier in states:
    row=subprocess.run(['docker','inspect','--format','{{.Id}}\t{{.Image}}\t{{.State.Status}}\t{{.State.Health.Status}}\t{{.RestartCount}}',identifier],capture_output=True,text=True,check=True,timeout=20).stdout.strip().split('\t')
    assert len(row)==5 and row[2:]==['running','healthy','0']
    current.append(row)
expected={c['id']:c['image'] for c in receipt['after']['containers']}
assert len(current)==9 and {r[0]:r[1] for r in current}==expected
assert stat.S_IMODE(Path('/home/deploy/rgnfarmasystemv2/.env').lstat().st_mode)==0o600
print(json.dumps({'checked_at':datetime.now(UTC).isoformat(),'status':'PASS','directory':str(DIR),'receipt_sha256':EXPECTED_RECEIPT,'artifact_hashes_verified':len(found),'files_mode0600':len(files),'rollback_tag':TAG,'rollback_image_id':image,'runtime_identities_unchanged_and_healthy':len(current),'checks':checks,'no_restore_or_restart':True,'postgres_format':'custom pg_dump; requires pg_restore, not the plain-SQL restore.sh path','limits':['hot backup','no joint DB/media atomicity','no isolated restoration','no offsite copy']},indent=2))
